{{- /* extend_footer.html --------------------------------------------------------------------------- Emits the fingerprinted JavaScript for Obsidian authoring enhancements. - js/link-preview.js only when the page has wikilinks or backlinks Every lookup goes through `with resources.Get`, so a missing asset degrades gracefully instead of failing the whole build. Idempotency: layouts/partials/footer.html invokes this partial through partialCached (keyed by layout/kind, not by page), so page-dependent script tags are emitted from the page template first, which sets the per-page "obsidianFooterScriptsDone" flag; the cached footer invocation then renders nothing. That keeps the preview script scoped to the pages that actually need it. There is deliberately no diagram script: diagrams are static SVG embedded with ![[diagram.svg|Alt text]] and need no JavaScript. */ -}} {{- if not (.Store.Get "obsidianFooterScriptsDone") -}} {{- $isBlogPost := and (eq .Kind "page") (eq .Section "blog") -}} {{- $needsPreview := or $isBlogPost (.Store.Get "hasWikilinks") (.Store.Get "hasBacklinks") -}} {{- if $needsPreview -}} {{- .Store.Set "obsidianFooterScriptsDone" true -}} {{- with resources.Get "js/link-preview.js" -}} {{- end -}} {{- end -}} {{- end -}}
