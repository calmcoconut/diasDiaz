{{- /* Returns a taxonomy term key's display name using the casing the author typed in post front matter ("OMSCS", "DnD") rather than the key Hugo derives internally. Hugo's own term title depends on `titleCaseStyle` and, for terms regenerated by an incremental build, can differ from a clean build; reading front matter keeps the tags index stable either way. */ -}} {{- $key := . -}} {{- $name := $key -}} {{- $found := false -}} {{- range site.RegularPages -}} {{- if not $found -}} {{- range .Params.tags -}} {{- if and (not $found) (eq (lower .) (lower $key)) -}} {{- $name = . -}} {{- $found = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $name -}}
